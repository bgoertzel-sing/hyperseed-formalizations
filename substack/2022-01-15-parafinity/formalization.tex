\begin{abstract}
We formalize Ben Goertzel's ``Parafinity'' (2022-01-15) within the
Hyperseed ontology. Key mappings: parafinity as the fiber boundary zone,
constructive nonstandard analysis as computable fiber, and AGI reasoning
as parafinite fiber operations.
\end{abstract}

\section{The parafinite zone: fiber boundary}

\begin{theorem}[Parafinity = fiber boundary --- claims 4--9, 13]
The parafinite zone is where fiber meets base --- too structured for raw
base treatment (finite enumeration) but not fully abstract (classical
infinity):
\[
  \text{finite} \subset \text{parafinite} \subset \text{infinite}
  \quad \text{with each } \subsetneq
\]
Parafinite structures have qualitatively different properties from both
ordinary finite and classical infinite --- they constitute a genuine
third zone, not just ``large finite'' or ``small infinite.''

In Hyperseed terms, fiber operations naturally live in the parafinite
zone: the fiber is too complex for finite enumeration of all fibers over
all base points, but structured enough (via the d-calculus) for
constructive manipulation.
\end{theorem}

\section{Computable fiber: constructive approach}

\begin{proposition}[Constructive = computable --- claims 1--3, 14]
Mycielski's constructive nonstandard analysis provides computable
infinitesimals --- infinitesimal-like objects that can actually be
computed with:
\[
  \epsilon_{\text{constructive}} \approx 0 \quad \text{but} \quad
  \text{computable}(\epsilon_{\text{constructive}}) = \text{true}
\]
This is essential for Hyperseed implementation: the fiber must be
constructively manipulable, not just abstractly defined. Computable
fiber is fiber with effective operations --- actual algorithms for
fiber transport, composition, and evolution.
\end{proposition}

\section{AGI in the parafinite zone}

\begin{conjecture}[AGI reasoning is parafinite --- claims 10--12, 15--16]
AGI reasoning naturally operates in the parafinite zone:
\[
  \text{AGI} \in \text{Parafinite}(\text{Fiber})
  = \{F : \text{finite} \subsetneq F \subsetneq \text{infinite}\}
\]
The d-calculus may be naturally parafinite --- operating on structures
too complex for finite enumeration but constructively manipulable.
Paraconsistency is especially natural in this zone, where classical
distinctions (finite/infinite, consistent/inconsistent) soften into
gradients.
\end{conjecture}

\begin{definition}[Resolution covering and parafiniteness --- claim 17]
\label{def:covering}
Let an observer carry a tolerance $\alpha$ on a set $B$ (symmetric,
$\alpha(x,x)=1$). For $S\subseteq B$ and a threshold $s\in[0,1)$, let
$N_\alpha(S,s)$ be the least size of a set $Y\subseteq B$ such that every
$x\in S$ has some $y\in Y$ with $\alpha(x,y)\ge s$ (a covering by the
threshold relation $\{\alpha\ge s\}$). Call $S$
\begin{itemize}
\item \emph{finite} if $|S|<\infty$;
\item \emph{parafinite} if $N_\alpha(S,s)<\infty$ for every $s<1$ and the map
  $s\mapsto$ (a covering of that size) is computable: finite at every
  resolution short of perfect;
\item \emph{wildly infinite} otherwise.
\end{itemize}
Every finite set is parafinite (cover by $S$ itself). The inclusion is strict:
$[0,1]$ with $\alpha(x,y)=1-|x-y|$ is parafinite and infinite. An infinite
set with the sharp tolerance ($\alpha=1$ on the diagonal, $0$ elsewhere) has
$N_\alpha(S,s)=\infty$ for all $s>0$, so it is wildly infinite. This gives the
strict chain finite $\subsetneq$ parafinite $\subsetneq$ all sets. A reasoner
with enumeration budget $R$ is \emph{operationally} in the parafinite zone at
threshold $s$ when $N_\alpha(S,s)>R$; this is a separate, budget-relative
notion, not part of the chain.
\end{definition}

\begin{proposition}[Computable approach along distinction --- claim 18]
\label{prop:dincrement}
Let $\bar\alpha^{\flat}=1-\alpha^{\flat}$ be the shortest-path pseudometric of
the metric closure. Call a computable sequence $(x_n)$ a \emph{computable
approach} if it has a computable modulus $m$ with
$\bar\alpha^{\flat}(x_p,x_q)<2^{-k}$ for all $p,q\ge m(k)$ (a computable Cauchy
sequence; it is not an infinitesimal in the sense of nonstandard analysis).
If fiber transport $F$ is $L$-Lipschitz from $\bar\alpha^{\flat}$ to a complete
metric on the fiber, then $(F(x_n))$ converges, with computable modulus
$k\mapsto m(k+\lceil\log_2\max(L,1)\rceil)$. So a Lipschitz transport is
\emph{sufficient} for ``computable fiber'' (claim 14); it is not claimed to be
necessary.
\end{proposition}

\begin{conjecture}[Paraconsistency lives on the sorites kernel --- claim 19]
\label{conj:subres}
The sorites kernel $\sigma=\alpha^{\flat}-\alpha\ge0$ is positive exactly on
pairs $(x,z)$ joined by a chain of steps each less distinct than the direct
comparison (the sorites paradox, Sec.\ \texttt{sec:transitivity} of the
paper). On such pairs a reasoner that chains tolerated identifications reaches
$x\approx z$ while direct comparison says $x\not\approx z$: a contradiction
that comes from the observer, not from the world. Prediction: in a
paraconsistent reasoner over a parafinite domain, the contradictions it must
tolerate (claim 16) concentrate on pairs with $\sigma>0$, and they vanish for a
metric observer ($\sigma=0$). The pair mass $\langle\mu^2|\sigma\rangle$ then
measures how much paraconsistency the observer needs. Untested.
\end{conjecture}
