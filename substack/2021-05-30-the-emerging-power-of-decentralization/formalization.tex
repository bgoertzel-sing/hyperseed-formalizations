\section{Distributed fiber bundles}

\begin{proposition}[Decentralization = distributed fiber --- claims 1--3, 6]
Decentralized systems are distributed fiber bundles:
\[
  F_{\text{decentralized}} = \bigoplus_{b \in B} F(b)
  \quad \text{(fiber distributed across base)}
\]
Rather than concentrating fiber at a single base point (centralized server,
institution, authority), decentralization distributes fiber across many
base points. SingularityNET distributes AI fiber across a blockchain network.
\end{proposition}

\section{Blockchain as fiber coordination}

\begin{proposition}[Blockchain = coordination protocol --- claims 2, 4, 7]
Blockchain is a coordination protocol for distributed fiber:
\[
  \text{consensus}(F(b_1), \ldots, F(b_n)) \implies \text{coherent bundle}
\]
The protocol ensures coherence of distributed fiber without centralized
control --- trustless coordination through mathematical/cryptographic
guarantees rather than institutional authority.
\end{proposition}

\section{Resilience as fiber redundancy}

\begin{proposition}[Resilience = fiber redundancy --- claims 5, 8]
Decentralized resilience comes from fiber redundancy:
\[
  \text{failure}(b_k) \implies F_{\text{total}} \approx F_{\text{total}} - F(b_k)
  \quad \text{(graceful degradation)}
\]
When fiber is redundantly distributed, loss of any single base point
doesn't destroy the overall fiber structure. No single point of failure.
\end{proposition}
