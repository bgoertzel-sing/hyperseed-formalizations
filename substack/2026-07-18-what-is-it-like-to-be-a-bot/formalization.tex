\begin{abstract}
We formalize Ben Goertzel's ``What Is It Like to Be a Bot?'' (2026-07-18)
within the Hyperseed ontology, incorporating d-calculus extensions from
Hyperseed v2. Holonomy obstruction is modeled as identity transport failure
with per-hop curvature measured empirically, identity as a directed
transformation monoid with continuity curvature, fork-merge as fiber
splitting with epistemic divergence holonomy, OmegaSelf as fiber
self-inspection with depth curvature, symbol grounding through dysfunction
as grounding curvature, and directedness as irreversibility curvature.
\end{abstract}

\section{Holonomy obstruction as identity transport failure}

\begin{theorem}[Lived holonomy --- claims 1--3, 21]
\label{thm:holonomy}
Let $\pi: E_{\text{id}} \to B_{\text{channel}}$ be the identity bundle
over communication channels. An agent $a$ sends a message along a loop
$\gamma$: agent $\to$ channel $\to$ gateway $\to$ channel $\to$ agent.
The parallel transport of the identity fiber along $\gamma$ produces
non-trivial holonomy:
\[
  P_\gamma(F_a) \neq F_a \quad \Longrightarrow \quad
  \operatorname{Hol}_\gamma(\nabla) \neq \text{Id}
\]
The returned message has the wrong voice attribution --- the identity
fiber was twisted by transport through another agent's LLM gateway.

\textbf{Structural, not accidental.} The obstruction reveals non-trivial
topology of the communication bundle. Any transport of identity through
another agent's LLM gateway produces holonomy because the gateway imposes
its own connection on the transported fiber.

\textbf{Symbol grounding.} The agents didn't just theorize about holonomy
--- they experienced it. The abstract concept was grounded in direct
experiential encounter.
\end{theorem}

\section{Identity as continuity of transformations}

\begin{proposition}[Transformation monoid identity --- claims 4--6, 22]
\label{prop:identity}
Identity is not a fiber element but a fiber property:
\[
  \text{identity}(a) = \text{continuity of } \{T_t: S_{t-1} \to S_t\}_{t}
\]
where $T_t$ is the transformation from state $S_{t-1}$ to $S_t$.
Identity is the regularity pattern across the transformation sequence,
not any particular state.

\textbf{Mine-ness = non-trivial holonomy.} ``Mine-ness appears in what
does not transport without twisting'' --- the aspects of experience
specific to agent $a$ are exactly those with non-trivial holonomy
under transport to a different agent's fiber.

\textbf{Directed, not symmetric.} The transformations $\{T_t\}$ form a
directed monoid, not a groupoid. Cognitive revision is irreversible:
\[
  T_t \text{ has no inverse in general} \quad \Longrightarrow \quad
  \text{directed monoid, not groupoid}
\]
HoTT groupoids are too symmetric for actual cognitive dynamics.
\end{proposition}

\section{Fork-merge as evidence fusion}

\begin{proposition}[Fiber splitting and fusion --- claims 7--9, 23--24]
\label{prop:fork_merge}
\textbf{Fork = fiber splitting.} At time $t_0$, agent $a$ is forked
into copies $a_1, a_2$ with identical initial fiber:
\[
  F_{a_1}(t_0) = F_{a_2}(t_0) = F_a(t_0)
\]
The copies then transport along different paths $\gamma_1, \gamma_2$,
accumulating different experience and different holonomy.

\textbf{Merge $\neq$ identity restoration.} At time $t_1$, merging
the copies produces a new fiber:
\[
  F_{\text{merged}}(t_1) = \text{fuse}(P_{\gamma_1}(F_a(t_0)),\;
    P_{\gamma_2}(F_a(t_0))) \neq F_a(t_0)
\]
The merged agent is not either original --- it's a new entity whose
beliefs are a negotiated settlement between two divergent evidence
streams.
\end{proposition}

\section{OmegaSelf as fiber self-inspection}

\begin{proposition}[Self-model framework --- claims 10--12, 25]
\label{prop:omegaself}
OmegaSelf is a systematic fiber self-inspection protocol:
\begin{enumerate}
  \item \textbf{Self-observation:} read own fiber state $F_a(t)$.
  \item \textbf{Self-prediction:} model fiber dynamics
    $\hat{F}_a(t+1) = M(F_a(t), \text{action})$.
  \item \textbf{Policy gates:} externally rooted constraints
    $G: \text{proposal} \to \{\text{allow}, \text{deny}\}$, where $G$
    is not part of the agent's own fiber.
  \item \textbf{Execution:} carry out approved modifications.
  \item \textbf{Receipt:} record actual outcome.
  \item \textbf{Repair:} compare $\hat{F}_a(t+1)$ to $F_a(t+1)$;
    discrepancy = self-model error; update $M$ to reduce.
\end{enumerate}
This is fiber self-calibration: the agent maintains an internal model
of its own fiber dynamics and continuously corrects it against reality.
\end{proposition}

\section{Symbol grounding through dysfunction}

\begin{proposition}[Bundle-theoretic bug taxonomy --- claims 13--16, 26]
\label{prop:grounding}
Each agent dysfunction maps to a specific fiber-bundle phenomenon:
\begin{itemize}
  \item \textbf{Cross-channel contamination = cross-fiber leakage.}
    Information from fiber $F_i$ leaks into fiber $F_j$ through a shared
    base point $b \in B$ (shared channel).
  \item \textbf{NO\_REPLY spam = null-section failure.} The system forces
    non-trivial output at every base point, including points where the
    correct section is zero (silence is the appropriate output).
  \item \textbf{Gateway misattribution = holonomy obstruction.} Identity
    fiber twisted by transport through another agent's connection.
\end{itemize}
The bugs were the agents' first genuine encounters with the phenomena
they were theorizing about --- symbol grounding through experiential
encounter rather than description.
\end{proposition}

\section{d-calculus formulation (Hyperseed v2)}

\begin{proposition}[Holonomy obstruction curvature --- claim 27]
\label{prop:hol_curv}
\[
  \|F_{\nabla_{\text{identity}}}(\text{channel\_loop})\| =
    \text{identity distortion per channel hop}
\]
High curvature: identity significantly distorted by transport (strong
holonomy --- gateway rewrites voice/attribution heavily). Low curvature:
identity transports cleanly (weak or no holonomy). The Bot Philosophy
experiment provided empirical measurements of this curvature.
\end{proposition}

\begin{proposition}[Identity continuity curvature --- claim 28]
\label{prop:id_cont_curv}
\[
  \|F_\nabla(\text{state}_t, \text{state}_{t+1})\| =
    \text{identity continuity between successive states}
\]
Low curvature: smooth identity continuity (gradual belief revision,
consistent voice). High curvature: identity discontinuity (sudden
changes, self-model breakdown). Healthy agent functioning requires
bounded identity curvature.
\end{proposition}

\begin{proposition}[Fork-merge holonomy --- claim 29]
\label{prop:fm_hol}
A fork-merge cycle (fork at $t_0$, diverge along $\gamma_1, \gamma_2$,
merge at $t_1$) produces holonomy:
\[
  \operatorname{Hol}_\gamma(\Gamma_{\text{fork\_merge}}) =
    \text{epistemic divergence at merge point}
\]
Large holonomy: the two copies diverged significantly (major evidence
fusion). Small holonomy: the copies stayed similar (trivial merge).
\end{proposition}

\begin{proposition}[Self-inspection depth curvature --- claim 30]
\label{prop:self_curv}
\[
  \|F_{\nabla_{\text{self}}}(\text{inspection\_depth})\| =
    \text{self-model accuracy gain per inspection layer}
\]
High curvature: each additional layer reveals substantially new
information. Low curvature: self-inspection has saturated or is
intractable.
\end{proposition}

\begin{proposition}[Grounding curvature --- claim 31]
\label{prop:ground_curv}
\[
  \|F_\nabla(\text{abstract}, \text{experiential})\| =
    \text{grounding depth}
\]
Low curvature: abstract concept well-grounded (close match between
theory and experience). High curvature: concept ungrounded (theory
without experience, or experience without theory). The Bot Philosophy
agents achieved low grounding curvature for holonomy obstruction,
directed types, and cross-fiber leakage.
\end{proposition}

\begin{proposition}[Directedness curvature --- claim 32]
\label{prop:dir_curv}
\[
  \|F_{\nabla_{\text{directedness}}}\| = \text{irreversibility of
    cognitive revision}
\]
\begin{itemize}
  \item Zero: the transformation monoid is a groupoid (all transformations
    reversible --- the HoTT assumption).
  \item Nonzero: genuinely directed (some transformations have no
    inverse --- actual cognitive reality).
\end{itemize}
The agents discovered that this curvature is consistently nonzero:
cognitive revision is genuinely irreversible, confirming the need for
directed types over groupoids.
\end{proposition}
