\section{Meta-crystallization as fiber emergence stack}

\begin{definition}[Emergence stack --- claims 3, 7]
The meta-crystallization stack is a fiber emergence stack with six
levels:
\begin{enumerate}
  \item[0.] Bare causal order $B_0$ (base space: discrete events with
    precedence).
  \item[1.] Occamistic selection fiber $E_1 \to B_0$ (simplicity bias
    selects regular patterns).
  \item[2.] Causal-PKC motifs $E_2 \to E_1$ (world-crystal structures
    emerge from selection).
  \item[3.] Meta-crystallized spacetime $E_3 \to E_2$ (continuous
    manifold from discrete crystal).
  \item[4.] Quaternionic dynamics $E_4 \to E_3$ (quantum theory and
    gauge fields over spacetime).
  \item[5.] Particles, fields, gravity $E_5 \to E_4$ (familiar physics
    as outermost fiber).
\end{enumerate}
Each level is a fiber over the level below: $E_{n+1} \to E_n$.
\end{definition}

\section{Occamistic selection: fiber-minimal selection}

\begin{proposition}[Fiber minimality --- claims 1, 6, 8]
\label{prop:occam}
Among all possible fiber structures over a given base, Occamistic
selection favors those with minimal representational cost:
\[
  E^* = \arg\min_{E \in \mathcal{E}} \text{complexity}(E)
    \quad \text{subject to } E \text{ supporting required dynamics}
\]
This is the same principle as quantale weakness in linguistics:
select the structure with least unnecessary fiber. Physics and
linguistics are shadows of the same selection principle applied
at different emergence levels.
\end{proposition}

\section{Wu Wei geodesics: fiber-minimal evolution}

\begin{proposition}[Minimal forcing --- claim 4, 9]
\label{prop:wuwei}
Wu Wei geodesics are paths of minimal fiber curvature:
\[
  \gamma^* = \arg\min_\gamma \int_\gamma \|F_{\nabla_\gamma}\|\, ds
\]
The system evolves along paths requiring least forcing. A true Wu Wei
geodesic has $\|F_{\nabla_\gamma}\| = 0$ everywhere along the path ---
pure inertial evolution with zero unnecessary forcing.
\end{proposition}

\section{d-calculus formulation (Hyperseed v2)}

\begin{proposition}[Emergence stack curvature --- claim 11]
\label{prop:emergence}
The curvature at each level measures the emergence gap:
\[
  \|F_\nabla(\text{level}_n, \text{level}_{n+1})\| =
    \text{emergence gap at level } n
\]
High curvature: qualitatively different next level (genuine emergence).
Low curvature: simple elaboration (mere complication). The stack has
high curvature at levels 0--1 (causal order to selection) and 2--3
(crystal to spacetime), indicating genuine emergence at those transitions.
\end{proposition}

\begin{proposition}[Occamistic curvature --- claim 12]
\label{prop:occam_curv}
\[
  \|F_{\nabla_\text{Occam}}\| = \text{strength of simplicity bias}
\]
Strong Occamistic curvature: only crystalline regularity survives
selection. Weak: amorphous variety survives. The world-crystal emerges
because the Occamistic curvature is strong enough to select crystal-like
structures from the vast space of possible causal sets.
\end{proposition}

\begin{proposition}[World-crystal holonomy --- claim 13]
\label{prop:holonomy}
A loop $\gamma$ in the world-crystal produces holonomy encoding
physical content:
\[
  \operatorname{Hol}_\gamma(\Gamma_\text{crystal}) = \text{gauge field
    strength enclosed by } \gamma
\]
This is not metaphorical: in gauge theory, field strength is literally
the holonomy of a connection. The d-calculus formulation embeds this
standard result within the emergence stack framework.
\end{proposition}

\begin{proposition}[Wu Wei geodesic curvature --- claim 14]
\label{prop:wuwei_curv}
\[
  \|F_{\nabla_\text{path}}\| = \text{forcing required along the path}
\]
True Wu Wei geodesic: $\|F_{\nabla_\text{path}}\| = 0$ (zero forcing,
pure inertial evolution). Real physical paths: small but nonzero forcing
curvature. The action principle of physics is: minimize
$\int \|F_{\nabla_\text{path}}\|\, ds$.
\end{proposition}

\begin{proposition}[Quaternionic gauge curvature --- claim 15]
