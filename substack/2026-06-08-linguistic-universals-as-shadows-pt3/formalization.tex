\section{Causal factorization as fiber decomposition}

\begin{theorem}[Cognitive substrate decomposition --- claims 1--5, 27]
\label{thm:factorization}
The cognitive substrate decomposes into fibers over different domains:
\[
  S_{\text{cognitive}} \cong \bigoplus_{d \in \text{Domains}} F_d
  \qquad \text{with} \qquad
  \text{Update}_A \text{ local to } F_{d(A)}
\]

Small commutators express approximate commutativity of fiber updates:
\[
  U_A \circ U_B \approx U_B \circ U_A
  \qquad \text{when } F_{d(A)} \cap F_{d(B)} = \emptyset
\]

Non-commutativity (catastrophic forgetting) is a fiber defect: the
commutator $[U_A, U_B]$ measures the defect. A well-factorized
architecture has small commutators for updates to unrelated domains.
Implementation-agnostic: what matters is relevant separability, not
the mechanism.
\end{theorem}

\section{Semantic frames as transition functions}

\begin{definition}[Cognition-to-language bundle --- claims 6--7, 28]
\label{def:frames}
Semantic frames are the transition functions of the cognition-to-language
bundle:
\[
  E_{\text{cog}} \xrightarrow{g_{\text{frame}}} E_{\text{lang}}
\]

The frame layer makes the bundle well-defined. Without it, the
cognition-to-language map is a naked map between incompatible coordinate
systems --- too rigid to do useful work. Frame types: event, force-dynamic,
argument-role, person/number, possession, attention.
\end{definition}

\section{Graded stability as fiber depth}

\begin{proposition}[Fiber depth --- claims 11, 14, 29]
\label{prop:depth}
Fiber depth encodes graded stability:
\begin{itemize}
  \item \textbf{Deep fiber (protected bedrock).} Person, number, case
    hierarchies. Slow learning rates. Survive across all languages.
  \item \textbf{Middle fiber (consolidated preferences).} Word-order
    regularities. Moderate protection.
  \item \textbf{Shallow fiber (plastic surface).} Context-bound universals.
    Fast learning rates. Most variation lives here.
\end{itemize}
The three layers of typological universals map directly to three fiber
depth levels.
\end{proposition}

\section{Training losses as fiber conditions}

\begin{proposition}[Naturality loss = cocycle condition --- claims 19, 30]
\label{prop:naturality}
The naturality loss enforces the cocycle condition: transition functions
compose consistently. Update-then-readout $\approx$ readout-then-
linguistic-update. The commuting diagram is the cocycle condition on the
cognition-to-language bundle.
\end{proposition}

\begin{proposition}[Closure loss = hierarchical cocycle --- claims 20, 32]
\label{prop:closure}
The closure loss enforces hierarchical consistency: fiber structure at
each level must be consistent with all levels below. If a language marks
a feature, it must mark all features below it on the hierarchy.
\end{proposition}

\begin{proposition}[Mediator loss = anti-scalar-collapse --- claims 21, 31]
\label{prop:mediator}
The mediator loss enforces anti-scalar-collapse: cross-module coupling
must be explicit and inspectable through a small number of dedicated
mediators, not diffuse and dense. Distant modules are conditionally
independent given the explicit mediator.
\end{proposition}

\begin{proposition}[Low-frustration loss = sparse compatibility --- claim 22]
\label{prop:frustration}
The low-frustration externalization loss enforces sparse compatibility:
local pulls from different modules must be compatible with a consistent
linearization over a sparse signed-energy graph.
\end{proposition}

\begin{proposition}[Stability schedule = fiber depth enforcement --- claim 23]
\label{prop:stability}
The stability schedule enforces the deep/shallow fiber distinction: slow
learning rates for deep fiber (protected bedrock), fast learning rates
for shallow fiber (plastic surface). This is direct enforcement of the
fiber depth structure.
\end{proposition}

\section{Linguistic typology as distributed fiber inspection}

\begin{proposition}[Distributed inspection --- claims 12--15, 33]
\label{prop:typology}
Each language externalizes a different cross-section of the cognitive
fiber. Typological comparison across languages is distributed fiber
inspection: many observers sampling the same underlying structure from
different vantage points.

Typological universals (features stable across unrelated languages) are
the fiber invariants --- the features of cognitive structure that survive
all externalization maps. Cross-cultural averaging (done by language
over thousands of years) is fiber extraction.
\end{proposition}

\section{Sparse routing and anti-scalar-collapse in coupling}
\label{sec:sparse}
%% =================================================================

\begin{proposition}[Explicit mediators --- claims 9--11, 23, 36]
\label{prop:sparse}
The article requires sparse externalization routing: connections
between cognitive and linguistic structure should be sparse --- a
small number of dedicated high-traffic pathways rather than dense
coupling everywhere. The evidence: the small number of broad
cross-module universals that survive genealogical and areal
correction (case marking, agreement morphology, argument-structure
resolution) are precisely the high-traffic mediators.

The mediator loss formalizes this: two distant module families should
be conditionally independent given an explicit mediator module. The
architecture is forbidden from implementing broad dependencies as
diffuse dense coupling between every relevant head; a small number of
explicit mediator heads must absorb that traffic.

In Hyperseed terms, this is \emph{anti-scalar-collapse applied to
inter-module coupling}: the coupling structure must be explicit and
inspectable, not collapsed into diffuse dense connections. Dense
coupling is scalar collapse of inter-module structure --- it hides
the dependency structure inside a matrix of small interactions
rather than exposing it as a small number of identifiable pathways.

The cost of dense coupling: interference, inference cost, and
catastrophic forgetting. These are the practical consequences of
scalar collapse in coupling --- when everything couples to everything,
there's no way to update one part without affecting all others, and
there's no way to inspect the dependency structure to understand why
a change propagated.
\end{proposition}

\begin{theorem}[Structural identity --- claims 28--30]
\label{thm:convergence}
The article's strongest claim: two independent lines of work ---
linguistic typology (empirical) and brain-like continual learning
(theoretical) --- converge on the same mathematical structure:

\begin{itemize}
\item Closure systems with diachronic Lyapunov flow.
\item Factorization theorems with mediator exceptions.
\item Sparse signed interaction graphs.
\item Graded stability under transition dynamics.
\item Gluing of context-indexed local closure operators.
\end{itemize}

This convergence is evidence that both lines are tracking the same
underlying phenomenon: the mathematical structure of cognition itself,
as it manifests both in the typological record (downstream linguistic
consequences) and in the design requirements for artificial cognitive
systems (upstream architectural constraints).

The Occam-like principle that picks out the right cognitive explanation
of a linguistic pattern is formalized as an algebraic order on
explanations combining five criteria: indistinction, fit, cost,
confusion, and double-counting. The result that causal explanations
beat correlation-only explanations ``falls out as a theorem in that
order'' --- it's not a preference but a consequence of the algebraic
structure.

In Hyperseed terms, this convergence is evidence for the Hyperseed
ontology itself: if two independent investigations of cognitive
architecture arrive at the same fiber-bundle structure (decomposition,
commutativity, transition functions, depth, cocycle conditions), that
structure is a genuine property of cognition, not an artifact of
either investigation's methodology.
\end{theorem}

\begin{conjecture}[Universality of the fiber-depth pattern]
Conjecture: the three-layer stability pattern (bedrock / consolidated
preferences / context-local residue) is not specific to language but
appears in every cognitive domain with sufficient developmental depth.
Visual perception (basic features / object categories / scene
interpretation), motor control (reflexes / skills / strategies),
and social cognition (basic emotion recognition / social norms /
cultural conventions) should all exhibit the same graded-stability
structure with the same mathematical properties (closure at the
deepest level, factorization at the middle level, sparse signed
interaction at the surface level).

If this conjecture holds, the linguistic universals training recipe
generalizes: any domain's ``universals'' (stable cross-context
features) can serve as training targets for the corresponding
cognitive module, using the same categorical loss functions with
domain-appropriate probes.
\end{conjecture}

\section{d-calculus formulation (Hyperseed v2)}

\begin{proposition}[Commutator curvature --- claim 34]
\label{prop:commutator_curv}
\[
  \|F_{\nabla_{\text{commutator}}}\| = \text{interference between
    unrelated domains}
\]
Zero curvature: perfect factorization (no interference, no catastrophic
forgetting). Nonzero curvature: imperfect factorization (interference
present). The training objective minimizes this curvature.
\end{proposition}

\begin{proposition}[Naturality holonomy --- claim 35]
\label{prop:naturality_hol}
A loop through the cognition-to-language bundle $\gamma$:
encode $\to$ frame $\to$ externalize $\to$ parse $\to$ decode produces
holonomy:
\[
  \operatorname{Hol}_\gamma(\Gamma_{\text{naturality}}) =
    \text{round-trip information loss}
\]
\begin{itemize}
  \item Trivial holonomy: perfect naturality (diagram commutes exactly,
    no information lost in round trip).
  \item Non-trivial holonomy: information loss (the diagram doesn't
    quite commute; some cognitive content is destroyed by the
    encode-decode cycle).
\end{itemize}
\end{proposition}

\begin{proposition}[Stability depth curvature --- claim 36]
\label{prop:depth_curv}
\[
  \|F_\nabla(\text{deep}, \text{shallow})\| = \text{protection boundary
    sharpness}
\]
High curvature: sharp boundary between protected bedrock and plastic
surface (good architecture; bedrock doesn't drift). Low curvature:
blurry boundary (unstable architecture; bedrock drifts, forgetting
creeps into core concepts).
\end{proposition}

\begin{proposition}[Frame bridge curvature --- claim 37]
\label{prop:frame_curv}
\[
  \|F_{\nabla_{\text{frame}}}\| = \text{frame translation fidelity}
\]
Low curvature: frames faithfully bridge cognitive and linguistic fiber
(clean translation, low loss). High curvature: frames distort the
translation (lossy bridge, cognitive content garbled in externalization).
\end{proposition}

\begin{proposition}[Typological inspection gradient --- claim 38]
\label{prop:typology_grad}
\[
  \nabla_{\text{typology}} = \nabla\left(\text{universality\_coefficient
    across language families}\right)
\]
Steep gradient: feature is universal (present in all families),
pointing toward cognitive bedrock. Flat gradient: feature is local
(family-specific), pointing toward surface variation. The gradient
direction identifies which cognitive fiber layers are deep (protected)
versus shallow (plastic).
\end{proposition}

Each term contributes a fiber-geometric force:
