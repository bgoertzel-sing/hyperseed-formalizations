\begin{abstract}
We formalize Ben Goertzel's ``What Is Science, Exactly?''
(2026-01-02) within the Hyperseed ontology, incorporating d-calculus
extensions from Hyperseed v2. Science is modeled as a communal fiber
bundle with structured quality control on the connection, paradigms as
connection regimes with discontinuous shifts, and Bayesian updating as
parallel transport. The reproducibility crisis is high curvature, and
AGI transforms science by steepening the evidence gradient.
\end{abstract}

\section{Science as communal fiber bundle}

\begin{definition}[Individual scientific fiber]
Each scientist $i$ has a fiber bundle $\pi_i: E_i \to B$ with fiber
encoding their knowledge, beliefs, methods, and theories.
\end{definition}

\begin{definition}[Communal fiber bundle --- claims 1, 2, 7]
The \emph{scientific community fiber bundle} is:
\[
  E_{\text{sci}} = \text{filtered-compose}(\{E_i\}_{i \in \text{community}})
\]
where the composition is filtered by structured skepticism: only fiber
elements that pass peer review, reproducibility checks, and communal
evaluation are accepted into $E_{\text{sci}}$.
\end{definition}

\begin{proposition}[Connection quality control --- claims 4, 8]
\label{prop:qc}
The scientific connection $\Gamma_{\text{sci}}$ filters fiber transport:
\[
  P_\gamma(\Gamma_{\text{sci}}): E_a \to E_b \quad \text{only if }
  \text{evidence}(\gamma) \text{ passes quality checks}
\]
Reproducibility is the primary quality check: fiber elements that can't
be independently reproduced are blocked from transport. Peer review
provides secondary quality control.
\end{proposition}

\section{Paradigm as connection regime}

\begin{definition}[Paradigm --- claim 5, 9]
A \emph{scientific paradigm} $\mathcal{P}$ is a connection regime ---
a specific connection $\Gamma_{\mathcal{P}}$ that determines how fiber
transports through the scientific community:
\[
  \mathcal{P} = (\Gamma_{\mathcal{P}}, \text{methods}_{\mathcal{P}},
    \text{questions}_{\mathcal{P}}, \text{standards}_{\mathcal{P}})
\]
Within a paradigm, transport is smooth and well-defined. Across paradigm
boundaries, transport is discontinuous or blocked.
\end{definition}

\begin{proposition}[Paradigm shift as discontinuous connection change]
\label{prop:shift}
A paradigm shift is:
\[
  \Gamma_{\mathcal{P}_{\text{old}}} \to \Gamma_{\mathcal{P}_{\text{new}}}
  \quad \text{(discontinuous)}
\]
The old connection is replaced entirely --- not smoothly deformed but
abruptly changed. Some fiber elements that transported well under the
old connection no longer transport under the new one, and vice versa.
\end{proposition}

\section{Bayesian updating as parallel transport}

\begin{proposition}[Evidence as parallel transport --- claims 3, 10]
\label{prop:bayes}
Bayesian updating on shared evidence is parallel transport in the
communal fiber bundle:
\begin{itemize}
  \item \textbf{Evidence} defines a path $\gamma$ in base space.
  \item \textbf{Prior} $P_0$ is the initial fiber state.
  \item \textbf{Posterior} $P_1 = P_\gamma(\Gamma_{\text{sci}}) \cdot P_0$
    is the fiber after transport.
\end{itemize}
The connection $\Gamma_{\text{sci}}$ encodes the methodology ---
different methodologies (connections) transport the same prior differently
along the same evidence path, producing different posteriors.
\end{proposition}

\section{Fiber revision: the Bayesian core}
\label{sec:bayesian}
%% =================================================================

\begin{proposition}[Bayesian fiber revision --- claims 10--13, 27]
\label{prop:bayesian}
At its best, science approximates Bayesian updating on a community-wide
prior. Each experiment revises the community fiber based on new base
evidence:
\[
  F_{t+1} = \text{Update}(F_t, e_{\text{new}}, w(e_{\text{new}}))
\]

where $w(e)$ is the evidential weight: controlled experiments weigh
more than observational studies, large samples more than small ones,
pre-registered studies more than post-hoc analyses.

The community prior evolves through accumulation of evidence, theory
development, and paradigm shifts. In Hyperseed terms, this is the
fiber's temporal evolution: the community fiber at time $t+1$ depends
on the fiber at time $t$ plus new base evidence.

Real science deviates from ideal Bayesianism in systematic ways:
confirmation bias (over-weighting evidence that confirms the current
fiber), anchoring (the prior fiber has too much inertia), social
pressure (the fiber reflects community politics, not just evidence),
and career incentives (the fiber reflects what's rewarded, not just
what's true).

These deviations are \emph{fiber distortions}: systematic ways in which
the community fiber fails to faithfully represent the base evidence.
Understanding these distortions is essential for improving the
fiber-construction process.
\end{proposition}

\section{Fiber-base decoupling: the reproducibility crisis}
\label{sec:crisis}
%% =================================================================

\begin{theorem}[Reproducibility crisis --- claims 14--18, 28]
\label{thm:crisis}
The current reproducibility crisis reveals what happens when the
social-epistemic machinery breaks down --- when the fiber (published
claims) decouples from the base (actual evidence):
\[
  \|F_{\text{published}} - F_{\text{evidence}}\| \gg 0
  \quad \text{(fiber-base decoupling)}
\]

The decoupling is driven by incentive misalignment:
\begin{itemize}
\item \textbf{Publish-or-perish}: rewards fiber quantity over fiber
  quality. The system incentivizes producing more fiber (papers)
  regardless of whether it faithfully represents the base (evidence).
\item \textbf{Publication bias}: journals preferentially accept fiber
  that shows positive results, creating a distorted community fiber.
  The ``file drawer problem'' means negative-result fiber disappears.
\item \textbf{Statistical malpractice}: p-hacking, HARKing, underpowered
  studies --- systematic methods for making the fiber \emph{appear} to
  represent the base when it doesn't.
\end{itemize}

The crisis extends beyond psychology to biomedical research, economics,
and other fields. It's a systemic fiber-base decoupling, not a
disciplinary one.

In Hyperseed terms, the reproducibility crisis is a failure of
\emph{fiber verification}: the community has stopped systematically
checking that the fiber faithfully represents the base. Without
verification (replication), the fiber floats free of its evidential
base --- claims accumulate without adequate base support.
\end{theorem}

\section{AI as base interface expansion for science}
\label{sec:ai}
%% =================================================================

\begin{proposition}[AI and science --- claims 19--24, 30]
\label{prop:ai}
AI could either help fix science or make it worse. In Hyperseed terms,
AI tools for science expand the base interface (faster analysis, more
data processing) but don't necessarily improve the fiber (better
theories, better reasoning):

\textbf{Helpful applications (base interface improvement):}
\begin{itemize}
\item Better statistical analysis (detecting fiber-base discrepancies).
\item Automated replication checking (systematic fiber verification).
\item Bias detection (identifying fiber distortions).
\item Literature synthesis (integrating partial fibers).
\end{itemize}

\textbf{Harmful applications (base interface abuse):}
\begin{itemize}
\item P-hacking at scale (automated fiber-base decoupling).
\item Publication mills (mass-produced unreliable fiber).
\item Deepfake data (corrupting the base itself).
\end{itemize}

AI is an amplifier: it amplifies whatever the system incentivizes. If
incentives are broken, AI amplifies the brokenness. This mirrors the
OpenClaw analysis: better hands for a scientific brain. The hands
(AI tools) amplify what the brain (scientific community) does, for
better or worse.

AGI-as-scientist requires genuine reasoning capabilities (abstraction,
causal inference, creative hypothesis generation) AND social integration
(participating in the communal epistemic process). This requires both
fiber improvement (better reasoning) and base integration (participating
in the community fiber construction).
\end{proposition}

\begin{conjecture}[Fiber verification as science's defining property]
Conjecture: the defining property of science (what distinguishes it from
other forms of collective belief) is systematic fiber verification ---
the commitment to checking that the communal fiber faithfully represents
the base evidence, through replication, peer review, and structured
skepticism.

If this conjecture holds, the reproducibility crisis is not a peripheral
problem but an existential threat to science itself: when fiber
verification breaks down, science ceases to be science and becomes
another form of collective myth-making. Restoring science requires
restoring fiber verification --- which means fixing the incentive
structures that currently discourage it.
\end{conjecture}

\section{d-calculus formulation (Hyperseed v2)}

\begin{proposition}[Reproducibility curvature --- claim 11]
\label{prop:reprod}
The curvature of the scientific connection measures irreproducibility:
\[
  \|F_{\nabla_{\text{sci}}}\| = \text{rate of irreproducibility}
\]
\begin{itemize}
  \item $\|F_{\nabla_{\text{sci}}}\| \ll 1$: results reproduce well
    across labs, contexts, and researchers (healthy science).
  \item $\|F_{\nabla_{\text{sci}}}\| \gg 1$: results fail to reproduce
    (reproducibility crisis --- degraded connection).
\end{itemize}
The ``reproducibility crisis'' in certain scientific domains is high
curvature: the connection has degraded, making parallel transport
path-dependent (different labs get different results from the same
evidence).
\end{proposition}

\begin{proposition}[Paradigm shift holonomy --- claim 12]
\label{prop:holonomy}
A paradigm shift produces maximal holonomy:
\[
  \|\operatorname{Hol}_\gamma(\Gamma_{\text{paradigm}}) - \mathrm{id}\|
  \text{ is maximal at paradigm shifts}
\]
The scientific community's fiber is maximally changed by traversing
a paradigm shift --- this is Kuhn's ``revolutionary science'' in
fiber-theoretic language.
\end{proposition}

\begin{proposition}[Evidence transport gradient --- claim 13]
\label{prop:gradient}
The evidence gradient in scientific fiber space:
\[
  \nabla_{\text{evidence}} = \nabla\left(\text{evidence\_weight} \times
    \text{reproducibility} \times \text{peer\_support}\right)
\]
Good science follows this gradient (toward well-supported conclusions);
bad science moves against it (p-hacking, publication bias, cherry-picking
move against $\nabla_{\text{evidence}}$).
\end{proposition}

\begin{proposition}[AGI steepens evidence gradient --- claims 6, 14]
\label{prop:agi}
AGI transforms science by steepening the evidence gradient:
\[
  \|\nabla_{\text{evidence}}\|_{\text{AGI}} \gg
    \|\nabla_{\text{evidence}}\|_{\text{human}}
\]
AGI can process more evidence, generate more hypotheses, and design
better experiments --- all of which increase the magnitude of
$\nabla_{\text{evidence}}$, making it easier to follow toward truth
and harder to move against toward error.
\end{proposition}
